{/* HSE TAB */}

                          {filteredIncidents.map((inc: any) => (

                                <span className="px-2 py-0.5 rounded-full text-[11px] font-bold bg-slate-100 dark:bg-slate-800 text-slate-700 dark:text-slate-300">
